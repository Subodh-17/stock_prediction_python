#   python3 -m venv .venv
#   . .venv/bin/activate

#   pip install -r webapp/requirements.txt

# requirements.txt REMAINS the default install path and is deliberately
# unpinned -- TensorFlow wheels in particular vary a lot across platforms and
# architectures, so hard pins can make a plain `pip install -r
